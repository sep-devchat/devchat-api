#include <iostream>
#include <vector>

int main() {
    std::vector<int> numbers = {5, 9, 2, 8};
    int sum = 0;

    for (int value : numbers) {
        sum += value;
    }

    double average = numbers.empty() ? 0.0 : static_cast<double>(sum) / numbers.size();
    std::cout << "Average: " << average << '\n';
    return 0;
}
